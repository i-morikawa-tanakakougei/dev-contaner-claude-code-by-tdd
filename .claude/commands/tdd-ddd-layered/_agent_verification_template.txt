# Template for Agent Result Verification section
3. **Agent Result Verification**:
   ```bash
   # 🔧 Load advanced task verification library
   source "$(dirname "${BASH_SOURCE[0]}")/_task_verification.sh"
   
   # 🔍 Verify agent execution results with advanced task checking
   echo "🔍 エージェント結果検証中..."
   
   verification_issues=()
   
   # Find latest report files
   latest_report=$(find_latest_report "COMMAND_NAME")
   
   if [[ -z "$latest_report" ]]; then
       verification_issues+=("レポートが生成されていません")
       echo "❌ レポートが見つかりません"
   else
       echo "  ✅ レポート: $latest_report"
       
       # ✨ Critical tasks verification
       echo "🔍 Critical tasks確認中..."
       if ! verify_critical_tasks "COMMAND_NAME" "$latest_report"; then
           echo "⚠️ Critical tasks確認で問題が検出されました"
           verification_issues+=("Critical tasks verification failed")
       else
           echo "✅ Critical tasks確認完了"
       fi
       
       # Basic content verification
       if ! grep -q "実行サマリー\|📊" "$latest_report"; then
           verification_issues+=("実行サマリーセクションが見つかりません")
       fi
       
       if ! grep -q "総合判定\|📋" "$latest_report"; then
           verification_issues+=("総合判定セクションが見つかりません")
       fi
   fi
   
   # Check agent exit code
   if [[ $agent_exit_code -ne 0 ]]; then
       verification_issues+=("エージェント実行エラー (終了コード: $agent_exit_code)")
   fi
   
   # ✨ Show comprehensive verification results
   show_verification_results "COMMAND_NAME"
   
   # Final decision based on verification results
   if [[ ${#verification_issues[@]} -eq 0 ]]; then
       echo "✅ エージェント結果検証完了"
   else
       echo "❌ エージェント結果検証で問題が発見されました:"
       printf '  - %s\n' "${verification_issues[@]}"
       echo ""
       echo "💡 問題解決のための推奨アクション:"
       echo "  1. 必要な前提条件の再確認"
       echo "  2. 適切なパラメータでのコマンド再実行"
       echo "  3. エラーログの詳細確認"
       exit 1
   fi
   ```
