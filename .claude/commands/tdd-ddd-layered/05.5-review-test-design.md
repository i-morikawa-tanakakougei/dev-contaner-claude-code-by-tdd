## Common Errors and Solutions

### ❌ Error Case 1: Tests are passing (not RED phase)
**Cause**: Tests were created with accidental implementation or are testing existing code  
**Solution**: 
```bash
# Verify tests fail properly
uv run --frozen pytest tests/ -v
# If tests pass, fix test implementation
/create-tests <issue-number>  # Recreate proper failing tests
# Then re-review
/review-test-design <issue-number>
```

### ❌ Error Case 2: Test files not found
**Cause**: Test review attempted before test creation  
**Solution**: 
```bash
# Create tests first
/create-tests <issue-number>
# Then proceed with review
/review-test-design <issue-number>
```

### ❌ Error Case 3: Incomplete Given-When-Then coverage
**Cause**: Tests don't cover all scenarios from specifications  
**Solution**: 
- Analyze missing scenarios and add corresponding tests
- Update test implementation to cover all acceptance criteria
- Ensure business edge cases are properly tested

   ```bash
   # 📋 Validate that test files exist

   missing_tests=()
   ```

3. **TDD RED Phase Validation**:
   ```bash
   # 🔴 Verify TDD RED phase compliance
   echo "🔴 TDD REDフェーズ検証中..."
   
   # Run tests to check if they fail properly
   test_results=$(uv run --frozen pytest tests/ -v --tb=short 2>&1 || echo "")
   
   # Count passed and failed tests
   passed_count=$(echo "$test_results" | grep -c "PASSED" || echo "0")
   failed_count=$(echo "$test_results" | grep -c "FAILED\|ERROR" || echo "0")
   total_tests=$((passed_count + failed_count))
   
   red_phase_violations=()
   red_phase_warnings=()
   
   if [[ $passed_count -gt 0 ]]; then
       red_phase_violations+=("${passed_count}個のテストが既に成功しています (TDD REDフェーズ違反)")
   fi
   
   if [[ $failed_count -eq 0 ]]; then
       red_phase_violations+=("失敗するテストが見つかりません")
   fi
   
   # Check failure reasons
   if echo "$test_results" | grep -q "ImportError\|ModuleNotFoundError"; then
       echo "  ✅ ImportError確認: 未実装モジュールによる正常な失敗"
   elif echo "$test_results" | grep -q "NotImplementedError"; then
       echo "  ✅ NotImplementedError確認: 未実装メソッドによる正常な失敗"
   else
       red_phase_warnings+=("失敗理由が不明確です - 実装が存在している可能性があります")
   fi
   
   # Report TDD RED phase status
   if [[ ${#red_phase_violations[@]} -eq 0 ]]; then
       echo "  ✅ 全テストが適切に失敗: ${failed_count}個のテスト"
       echo "  ✅ TDD REDフェーズ確認済み"
   else
       echo "  ❌ TDD REDフェーズ違反が検出されました:"
       printf '    ⚠️ %s\n' "${red_phase_violations[@]}"
   fi
   
   if [[ ${#red_phase_warnings[@]} -gt 0 ]]; then
       echo "  ⚠️ TDD警告事項:"
       printf '    💡 %s\n' "${red_phase_warnings[@]}"
   fi
   ```

4. **Given-When-Then Coverage Analysis**:
   ```bash
   # 📊 Analyze Given-When-Then scenario coverage
   echo "📊 Given-When-Thenカバレッジ分析中..."
   
   coverage_issues=()
   coverage_warnings=()
   
   # Find use case specification files
   spec_files=()
   for issue_num in "${issue_numbers[@]}"; do
       spec_file=$(find docs/use_cases/ -name "*issue-*${issue_num}*" -type f -name "*.md" | head -1)
       if [[ -n "$spec_file" ]]; then
           spec_files+=("$spec_file")
       fi
   done
   
   if [[ ${#spec_files[@]} -gt 0 ]]; then
       # Extract scenarios from specifications
       total_scenarios=0
       covered_scenarios=0
       
       for spec_file in "${spec_files[@]}"; do
           # Count Given-When-Then scenarios
           scenarios=$(grep -c "Given:\|When:\|Then:" "$spec_file" 2>/dev/null || echo "0")
           total_scenarios=$((total_scenarios + scenarios / 3))  # Assuming each scenario has all three parts
           
           # Check if scenarios are covered in tests
           scenario_patterns=$(grep -E "Given:|When:|Then:" "$spec_file" 2>/dev/null | head -15 || echo "")
           if [[ -n "$scenario_patterns" ]]; then
               while IFS= read -r pattern; do
                   # Look for related test methods
                   test_pattern=$(echo "$pattern" | sed 's/Given:\|When:\|Then://g' | tr '[:upper:]' '[:lower:]' | sed 's/[^a-z0-9]/_/g')
                   if find tests/ -name "*.py" -exec grep -l "$test_pattern\|test.*$(echo "$test_pattern" | cut -c1-10)" {} \; 2>/dev/null | head -1 >/dev/null; then
                       covered_scenarios=$((covered_scenarios + 1))
                   fi
               done <<< "$scenario_patterns"
           fi
       done
       
       # Calculate coverage percentage
       if [[ $total_scenarios -gt 0 ]]; then
           coverage_percent=$(( (covered_scenarios * 100) / total_scenarios ))
           echo "  📊 シナリオカバレッジ: ${coverage_percent}% (${covered_scenarios}/${total_scenarios}シナリオ)"
           
           if [[ $coverage_percent -ge 90 ]]; then
               echo "  ✅ 受け入れ基準カバレッジ: 良好"
           elif [[ $coverage_percent -ge 70 ]]; then
               coverage_warnings+=("シナリオカバレッジが不十分: ${coverage_percent}%")
           else
               coverage_issues+=("シナリオカバレッジが低すぎます: ${coverage_percent}%")
           fi
       else
           coverage_warnings+=("Given-When-Thenシナリオが仕様書で見つかりません")
       fi
   else
       coverage_warnings+=("ユースケース仕様ファイルが見つかりません")
   fi
   
   # Report coverage status
   if [[ ${#coverage_issues[@]} -eq 0 ]]; then
       echo "  ✅ Given-When-Thenカバレッジ: 適切"
   else
       echo "  ❌ カバレッジ問題が検出されました:"
       printf '    ⚠️ %s\n' "${coverage_issues[@]}"
   fi
   
   if [[ ${#coverage_warnings[@]} -gt 0 ]]; then
       echo "  ⚠️ カバレッジ改善推奨事項:"
       printf '    💡 %s\n' "${coverage_warnings[@]}"
   fi
   ```

5. **Test Quality Assessment**:
   ```bash
   # 🧪 Assess test design quality
   echo "🧪 テスト品質評価中..."
   
   quality_issues=()
   quality_warnings=()
   
   # Analyze test structure
   unit_tests=$(find tests/ -path "*/unit/*" -name "*.py" | wc -l)
   integration_tests=$(find tests/ -path "*/integration/*" -name "*.py" | wc -l)
   e2e_tests=$(find tests/ -path "*/e2e/*" -name "*.py" | wc -l)
   
   echo "  📊 テスト構造分析:"
   echo "    - ユニットテスト: ${unit_tests}ファイル"
   echo "    - 統合テスト: ${integration_tests}ファイル"
   echo "    - E2Eテスト: ${e2e_tests}ファイル"
   
   # Check test naming conventions
   poorly_named_tests=0
   well_named_tests=0
   
   for test_file in "${unique_test_files[@]}"; do
       # Count test methods
       test_methods=$(grep -c "def test_" "$test_file" 2>/dev/null || echo "0")
       
       # Check for descriptive test names
       descriptive_names=$(grep "def test_" "$test_file" 2>/dev/null | grep -c "_when_\|_should_\|_given_\|_then_" || echo "0")
       
       if [[ $descriptive_names -gt 0 ]] && [[ $test_methods -gt 0 ]]; then
           if [[ $((descriptive_names * 100 / test_methods)) -ge 70 ]]; then
               well_named_tests=$((well_named_tests + descriptive_names))
           else
               poorly_named_tests=$((poorly_named_tests + test_methods - descriptive_names))
           fi
       else
           poorly_named_tests=$((poorly_named_tests + test_methods))
       fi
   done
   
   # Report quality assessment
   if [[ $unit_tests -gt 0 ]] && [[ $integration_tests -gt 0 ]]; then
       echo "  ✅ テスト構造: 適切"
   else
       quality_warnings+=("テスト構造の改善推奨: ユニット/統合テストの分離")
   fi
   
   if [[ $well_named_tests -gt $poorly_named_tests ]]; then
       echo "  ✅ テスト命名: ビジネス意図が明確"
   else
       quality_warnings+=("テスト命名の改善推奨: ビジネス意図を明確に")
   fi
   
   # Simple assertion quality check
   assertion_count=0
   for test_file in "${unique_test_files[@]}"; do
       assertions=$(grep -c "assert\|assertEqual\|assertTrue\|assertFalse" "$test_file" 2>/dev/null || echo "0")
       assertion_count=$((assertion_count + assertions))
   done
   
   if [[ $assertion_count -gt 0 ]]; then
       echo "  ✅ アサーション品質: 良好"
   else
       quality_issues+=("アサーションが見つかりません")
   fi
   
   # Report quality status
   if [[ ${#quality_issues[@]} -eq 0 ]]; then
       echo "  ✅ テスト品質: 基準を満たしています"
   else
       echo "  ❌ テスト品質問題が検出されました:"
       printf '    ⚠️ %s\n' "${quality_issues[@]}"
   fi
   
   if [[ ${#quality_warnings[@]} -gt 0 ]]; then
       echo "  ⚠️ テスト品質改善推奨事項:"
       printf '    💡 %s\n' "${quality_warnings[@]}"
   fi
   ```

   ```bash
   for issue_num in "${issue_numbers[@]}"; do
       metadata_file=$(find docs/use_cases -name "issue-${issue_num}-*.json" | head -1)
       if [[ -f "$metadata_file" ]]; then
           # Update test review status in metadata
           update_metadata_atomic "$metadata_file" \
               --set ".phases.test_design_review.reviewed = true" \
               --set ".phases.test_design_review.review_status = \"$review_status\"" \
               --set ".phases.test_design_review.review_date = \"$(date -Iseconds)\"" \
               --set ".phases.test_design_review.review_report = \"$review_report\"" \
               --set ".phases.test_design_review.test_count = $total_tests" \
               --set ".phases.test_design_review.failing_tests = $failed_count"
       fi
   done
   
   # Provide final decision and next steps
   echo ""
   echo "$status_icon テスト設計レビュー結果: $review_status"
   
   case "$review_status" in
       "APPROVED")
           echo "✅ テスト設計承認: 実装開始可能"
           echo "💡 次のステップ: /implement-domain ${issue_numbers[*]}"
           ;;
       "CONDITIONAL_APPROVAL")
           echo "⚠️ 条件付き承認: 軽微な改善推奨"
           echo "💡 次のステップ: 改善適用後に /implement-domain ${issue_numbers[*]}"
           ;;
       "REJECTED")
           echo "❌ テスト修正が必要です"
           echo "💡 次のステップ: /create-tests ${issue_numbers[*]} (テスト再作成)"
           echo ""
           echo "主な修正事項:"
           printf '  - %s\n' "${red_phase_violations[@]}" "${coverage_issues[@]}" "${quality_issues[@]}"
           exit 1
           ;;
   esac
   
   echo "🎉 テスト設計レビュー完了!"
   ```
