## Common Errors and Solutions

### ❌ Error Case 1: Test execution results not found
**Cause**: Test results review attempted before test execution  
**Solution**: 
```bash
# Execute tests first
/run-all-tests <issue-number>
# Then proceed with review
/review-test-results <issue-number>
```

### ❌ Error Case 2: Low test coverage detected
**Cause**: Implementation doesn't have adequate test coverage  
**Solution**: 
- Identify uncovered code paths and add tests
- Focus on critical business logic coverage
- Ensure Given-When-Then scenarios are fully tested

### ❌ Error Case 3: Performance issues detected
**Cause**: Implementation has performance bottlenecks  
**Solution**: 
- Analyze slow tests and optimize critical paths
- Review database queries and external dependencies
- Consider caching strategies for expensive operations

2. **Validate Prerequisites and Gather Test Results**:
   ```bash
   # 📊 Validate that test execution results exist
   echo "📊 テスト実行結果分析中..."
   
   # Check if tests directory exists
   if [[ ! -d "tests/" ]]; then
       echo "❌ テストディレクトリが見つかりません"
       echo "💡 最初に以下のコマンドでテストを実行してください:"
       echo "   /run-all-tests ${issue_numbers[*]}"
       exit 1
   fi
   
   # Run tests to get current results
   echo "  🧪 テスト実行中..."
   test_output=$(uv run --frozen pytest tests/ -v --tb=short 2>&1 || echo "")
   
   # Extract test metrics
   total_tests=$(echo "$test_output" | grep -E "=+ .* =+$" | grep -oE "[0-9]+ passed|[0-9]+ failed|[0-9]+ error" | awk '{sum += $1} END {print sum}')
   passed_tests=$(echo "$test_output" | grep -oE "[0-9]+ passed" | grep -oE "[0-9]+" || echo "0")
   failed_tests=$(echo "$test_output" | grep -oE "[0-9]+ failed" | grep -oE "[0-9]+" || echo "0")
   error_tests=$(echo "$test_output" | grep -oE "[0-9]+ error" | grep -oE "[0-9]+" || echo "0")
   
   # Calculate success rate
   if [[ $total_tests -gt 0 ]]; then
       success_rate=$(( (passed_tests * 100) / total_tests ))
       echo "  📈 テスト成功率: ${success_rate}% (${passed_tests}/${total_tests}テスト)"
   else
       echo "  ❌ テスト結果が取得できません"
       echo "💡 テストを実行してください: /run-all-tests ${issue_numbers[*]}"
       exit 1
   fi
   
   # Get execution time
   execution_time=$(echo "$test_output" | grep -oE "in [0-9]+\.[0-9]+s" | grep -oE "[0-9]+\.[0-9]+" || echo "unknown")
   echo "  ⏱️ 実行時間: ${execution_time}秒"
   ```

3. **Coverage Analysis**:
   ```bash
   # 📊 Analyze test coverage
   echo "📊 カバレッジ分析中..."
   
   coverage_issues=()
   coverage_warnings=()
   
   # Generate coverage report
   echo "  📈 カバレッジレポート生成中..."
   coverage_output=$(uv run --frozen pytest --cov=src --cov-report=term 2>&1 || echo "")
   
   # Extract coverage percentage
   if echo "$coverage_output" | grep -q "TOTAL"; then
       total_coverage=$(echo "$coverage_output" | grep "TOTAL" | awk '{print $4}' | sed 's/%//')
       echo "  📊 カバレッジ分析: ${total_coverage}% (target: 80%+)"
       
       if [[ $total_coverage -ge 80 ]]; then
           echo "  ✅ カバレッジ目標達成"
       elif [[ $total_coverage -ge 70 ]]; then
           coverage_warnings+=("カバレッジ改善推奨: ${total_coverage}%")
       else
           coverage_issues+=("カバレッジ不足: ${total_coverage}% (target: 80%+)")
       fi
   else
       coverage_warnings+=("カバレッジ情報を取得できませんでした")
   fi
   
   # Analyze coverage by layer
   layers=("domain" "application" "infrastructure" "presentation")
   for layer in "${layers[@]}"; do
       if [[ -d "src/$layer" ]]; then
           layer_coverage=$(echo "$coverage_output" | grep "src/$layer" | awk '{print $4}' | sed 's/%//' | head -1)
           if [[ -n "$layer_coverage" ]]; then
               echo "    - ${layer}層: ${layer_coverage}%"
               if [[ $layer_coverage -lt 70 ]]; then
                   coverage_warnings+=("${layer}層のカバレッジが低いです: ${layer_coverage}%")
               fi
           fi
       fi
   done
   
   # Report coverage status
   if [[ ${#coverage_issues[@]} -eq 0 ]]; then
       echo "  ✅ カバレッジ分析: 良好"
   else
       echo "  ❌ カバレッジ問題が検出されました:"
       printf '    ⚠️ %s\n' "${coverage_issues[@]}"
   fi
   
   if [[ ${#coverage_warnings[@]} -gt 0 ]]; then
       echo "  ⚠️ カバレッジ改善推奨事項:"
       printf '    💡 %s\n' "${coverage_warnings[@]}"
   fi
   ```

4. **Given-When-Then Scenario Validation**:
   ```bash
   # 🎯 Validate Given-When-Then scenario implementation
   echo "🎯 Given-When-Thenカバレッジ確認中..."
   
   scenario_issues=()
   scenario_warnings=()
   
   # Find use case specification files
   spec_files=()
   for issue_num in "${issue_numbers[@]}"; do
       spec_file=$(find docs/use_cases/ -name "*issue-*${issue_num}*" -type f -name "*.md" | head -1)
       if [[ -n "$spec_file" ]]; then
           spec_files+=("$spec_file")
       fi
   done
   
   if [[ ${#spec_files[@]} -gt 0 ]]; then
       total_scenarios=0
       implemented_scenarios=0
       
       for spec_file in "${spec_files[@]}"; do
           # Count Given-When-Then scenarios
           gwt_count=$(grep -c "Given:\|When:\|Then:" "$spec_file" 2>/dev/null || echo "0")
           scenario_count=$((gwt_count / 3))  # Assuming each scenario has all three parts
           total_scenarios=$((total_scenarios + scenario_count))
           
           # Check if scenarios are implemented in tests
           scenario_text=$(grep -A 2 -B 2 "Given:\|When:\|Then:" "$spec_file" 2>/dev/null || echo "")
           while IFS= read -r line; do
               if [[ -n "$line" ]]; then
                   # Look for test implementation
                   test_search=$(echo "$line" | sed 's/Given:\|When:\|Then://g' | tr '[:upper:]' '[:lower:]' | sed 's/[^a-z0-9]/_/g' | cut -c1-15)
                   if find tests/ -name "*.py" -exec grep -l "$test_search\|test.*$(echo "$test_search" | cut -c1-8)" {} \; 2>/dev/null | head -1 >/dev/null; then
                       implemented_scenarios=$((implemented_scenarios + 1))
                   fi
               fi
           done <<< "$scenario_text"
       done
       
       # Calculate implementation rate
       if [[ $total_scenarios -gt 0 ]]; then
           implementation_rate=$(( (implemented_scenarios * 100) / total_scenarios ))
           echo "  📊 シナリオ実装率: ${implementation_rate}% (${implemented_scenarios}/${total_scenarios}シナリオ)"
           
           if [[ $implementation_rate -ge 90 ]]; then
               echo "  ✅ 受け入れ基準: すべて満たしている"
           elif [[ $implementation_rate -ge 80 ]]; then
               scenario_warnings+=("一部シナリオの実装が不完全: ${implementation_rate}%")
           else
               scenario_issues+=("シナリオ実装率が不足: ${implementation_rate}%")
           fi
       else
           scenario_warnings+=("Given-When-Thenシナリオが特定できません")
       fi
   else
       scenario_warnings+=("ユースケース仕様ファイルが見つかりません")
   fi
   
   # Report scenario validation status
   if [[ ${#scenario_issues[@]} -eq 0 ]]; then
       echo "  ✅ シナリオ実装: 完了"
   else
       echo "  ❌ シナリオ実装問題が検出されました:"
       printf '    ⚠️ %s\n' "${scenario_issues[@]}"
   fi
   
   if [[ ${#scenario_warnings[@]} -gt 0 ]]; then
       echo "  ⚠️ シナリオ実装改善推奨事項:"
       printf '    💡 %s\n' "${scenario_warnings[@]}"
   fi
   ```

       if [[ $complex_methods -eq 0 ]]; then
           echo "  ✅ 循環複雑度: 良好"

   arch_violations=0

       arch_violations=$((arch_violations + domain_violations))
   fi
   
   if [[ $arch_violations -eq 0 ]]; then
       echo "  ✅ アーキテクチャ準拠: DDD/Clean違反なし"
   else
       quality_issues+=("アーキテクチャ違反: ${arch_violations}件")
   fi
   
   # Report quality metrics status
   if [[ ${#quality_issues[@]} -eq 0 ]]; then
       echo "  ✅ 品質メトリクス: 良好"
   else
       echo "  ❌ 品質問題が検出されました:"

   fi
   
   if [[ ${#quality_warnings[@]} -gt 0 ]]; then
       echo "  ⚠️ 品質改善推奨事項:"

   fi
