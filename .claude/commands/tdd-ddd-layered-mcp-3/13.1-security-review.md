        Use mcp__context7__get-library-docs "/domain-driven-design" --topic "business-rule-security" --tokens=1500
    fi
    
    echo "🛡️ Analyzing Domain Layer security patterns..."
    echo "  Focus: Business rule security, entity validation, domain invariants, data integrity"
    
    # Domain セキュリティパターン分析
    if command -v uv &> /dev/null; then
        echo "🔬 Analyzing Domain layer security implementations..."
        
        # Domain セキュリティ関連テスト実行
        if [[ -d "tests/domain" ]]; then
            Use Bash tool: uv run --frozen pytest tests/domain/ -v --tb=short -k "security or validation or auth"
            DOMAIN_SECURITY_RESULT=$?
        elif [[ -d "tests/unit/domain" ]]; then
            Use Bash tool: uv run --frozen pytest tests/unit/domain/ -v --tb=short -k "security or validation or auth"
            DOMAIN_SECURITY_RESULT=$?
        else
            echo "⚠️ Domain test directory not found - checking for domain security tests"
            Use Bash tool: uv run --frozen pytest -k "domain and (security or validation or auth)" -v --tb=short
            DOMAIN_SECURITY_RESULT=$?
        fi
        
        if [[ $DOMAIN_SECURITY_RESULT -eq 0 ]]; then
            echo "✅ SECURE: Domain Layer security validation passed"
            echo "📋 EVIDENCE: Business rule security and data validation confirmed"
            echo "🛡️ SECURITY STATUS: Domain layer security VALIDATED"
            return 0
        else
            echo "⚠️ SECURITY RISK: Domain Layer has potential security issues"
            echo "📋 EVIDENCE: Business rule validation or data integrity issues detected"  
            echo "🎯 ACTION REQUIRED: Domain layer security review and hardening needed"
            return 1
        fi
    else
        echo "⚠️ Test runner not available - providing manual security validation guidance"
        echo "💡 Manually verify: Business rule validation, entity invariants, data integrity"
        return 2
    fi
}

implement_mcp3_security_remediation() {
    echo "🛡️ MCP-3 Intelligent Security Remediation Strategy..."
    
    local security_risk_detected="false"
    local risk_layer=""
    local risk_type=""
    local risk_level=""
    
    # セキュリティリスク層の特定
    if [[ "${DOMAIN_SECURITY_RESULT:-0}" -ne 0 ]]; then
        security_risk_detected="true"
        risk_layer="DOMAIN"
        risk_type="BUSINESS_LOGIC_SECURITY"
        risk_level="HIGH"
        echo "🎯 Security Risk Strategy: Domain business logic security hardening"
    elif [[ "${USECASE_SECURITY_RESULT:-0}" -ne 0 ]]; then
        security_risk_detected="true"
        risk_layer="USECASE"  
        risk_type="AUTHENTICATION_AUTHORIZATION"
        risk_level="HIGH"
        echo "🎯 Security Risk Strategy: Use case authentication/authorization hardening"
    elif [[ "${INFRA_SECURITY_RESULT:-0}" -ne 0 ]]; then
        security_risk_detected="true"
        risk_layer="INFRASTRUCTURE"
        risk_type="DATA_PROTECTION"
        risk_level="HIGH"
        echo "🎯 Security Risk Strategy: Infrastructure data protection and encryption hardening"
    elif [[ "${PRESENTATION_SECURITY_RESULT:-0}" -ne 0 ]]; then
        security_risk_detected="true"
        risk_layer="PRESENTATION"
        risk_type="INPUT_OUTPUT_SECURITY"
        risk_level="HIGH"
        echo "🎯 Security Risk Strategy: Presentation input validation and output encoding hardening"
    else
        echo "✅ No layer-specific security risks detected - all layers security validated"
        return 0
    fi
    
    if [[ "$security_risk_detected" == "true" ]]; then
        echo "🛡️ Implementing systematic security remediation for: $risk_layer"
        
        if [[ "$MCP3_SECURITY_MODE" == "ENHANCED" ]]; then
            echo "🧠 Enhanced Security Strategy with MCP-3 Intelligence..."
            
            # Context7セキュリティ修正パターン適用
            echo "📚 Context7: Retrieving security remediation patterns and compliance standards..."
            Use mcp__context7__get-library-docs "/security-patterns" --topic "vulnerability-remediation" --tokens=2500
            Use mcp__context7__get-library-docs "/owasp-guidelines" --topic "security-hardening" --tokens=2000
            
            # Serena影響分析・最適修正点特定  
            echo "🔍 Serena: Security impact analysis and optimal remediation point identification..."
            Use mcp__serena__find_referencing_symbols for security impact assessment
            Use mcp__serena__search_for_pattern "SECURITY|VULNERABILITY|FIXME.*security|TODO.*security" --context_lines_before=2 --context_lines_after=5
            
            # 🆘 セキュリティリスクが複雑な場合の追加分析手法
            echo "🔬 Additional security analysis for complex vulnerabilities..."
            
            # セキュリティログ・監査証跡の追記
            echo "📝 Adding security logging for vulnerability identification..."
            Use mcp__serena__search_for_pattern "audit|log.*security|security.*log|track.*security" --context_lines_before=3 --context_lines_after=3
            echo "💡 Consider adding security audit logging at critical security checkpoints"
            
            # Context7による最新脆弱性対策Web検索
            echo "🌐 Searching for latest vulnerability remediation in knowledge base..."
            Use mcp__context7__resolve-library-id "${risk_layer,,}-security-remediation"
            Use mcp__context7__resolve-library-id "common-vulnerability-fixes"
            
            # 具体的なセキュリティリスクによる対策検索
            if [[ "$risk_layer" == "DOMAIN" ]]; then
                Use mcp__context7__get-library-docs "/domain-security-remediation" --topic "business-logic-vulnerabilities" --tokens=3000
            elif [[ "$risk_layer" == "USECASE" ]]; then
                Use mcp__context7__get-library-docs "/usecase-security-remediation" --topic "authentication-failures" --tokens=3000
            elif [[ "$risk_layer" == "INFRASTRUCTURE" ]]; then
                Use mcp__context7__get-library-docs "/infrastructure-security-remediation" --topic "data-protection-vulnerabilities" --tokens=3000
            elif [[ "$risk_layer" == "PRESENTATION" ]]; then
                Use mcp__context7__get-library-docs "/presentation-security-remediation" --topic "input-validation-vulnerabilities" --tokens=3000
            fi
            
            # Web検索による最新セキュリティ対策事例の取得
            echo "🔍 Searching web for latest security remediation solutions..."
            Use WebSearch to find solutions for: "${risk_layer,,} layer security vulnerabilities ${risk_type,,} remediation techniques"
            Use WebSearch to find solutions for: "OWASP ${risk_layer,,} security hardening best practices"
        fi
        
        # ユーザーにセキュリティ修正実装を依頼（日本語）
        Ask user in Japanese: 
        "🛡️ MCP-3セキュリティ分析に基づく修正が必要です：
        
        **セキュリティリスク層**: $risk_layer
        **リスクタイプ**: $risk_type
        **リスクレベル**: $risk_level
        **証拠**: 層別セキュリティテストによる明確なリスク特定完了
        
        MCP-3セキュリティ分析結果に基づいて、以下の修正アプローチを推奨します：
        1. 特定されたセキュリティリスク層のみに集中した最小限修正
        2. 他層への影響を最小化するセキュリティ強化アプローチ  
        3. 修正後の段階的セキュリティ検証と回帰テスト実行
        
        🆘 **セキュリティリスクが複雑な場合の追加手法**:
        - セキュリティ監査ログを重要なチェックポイントに追加
        - Context7から最新の脆弱性対策パターンを取得済み
        - Web検索により類似セキュリティ問題の対策事例を調査済み
        - 段階的なセキュリティテストケース追加による問題の切り分け
        
        セキュリティ強化を実装しますか？(y/N)"
        
        # セキュリティ修正実装の指導
        if user confirms security fix implementation; then
            echo "🔨 Implementing guided security fix with MCP-3 recommendations..."
            
            # 修正ガイダンス提供
            Provide specific security fix guidance based on:
            - Layer-specific security issues identified
            - MCP-3 security analysis results
            - Context7 security best practices
            - Minimal impact security hardening approach
            
            # 修正後セキュリティ検証
            echo "✅ Security fix implementation completed - validating security improvements..."
            
            # 修正された層のセキュリティテスト再実行
            case "$risk_layer" in
                "DOMAIN")
                    execute_domain_security_analysis
                    ;;
                "USECASE")
                    execute_usecase_security_analysis
                    ;;
                "INFRASTRUCTURE") 
                    execute_infra_security_analysis
                    ;;
                "PRESENTATION")
                    execute_presentation_security_analysis
                    ;;
            esac
            
            if [[ $? -eq 0 ]]; then
                echo "✅ Security validation successful - vulnerability resolved"
                return 0
            else
                echo "❌ Security validation failed - additional analysis required"
                return 1
            fi
        else
            echo "ℹ️ Security fix implementation deferred - security analysis completed"
            return 0
        fi
    fi
}
