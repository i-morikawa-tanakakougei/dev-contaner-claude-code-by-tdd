            echo "💡 Consider adding debug prints/logs at key points to trace execution flow"

            echo "🌐 Searching for solutions in latest knowledge base..."
            Use mcp__context7__resolve-library-id "${fix_layer,,}-debugging-solutions"
            Use mcp__context7__resolve-library-id "common-integration-errors"

            if [[ "$fix_layer" == "DOMAIN" ]]; then
                Use mcp__context7__get-library-docs "/domain-debugging-solutions" --topic "business-logic-errors" --tokens=3000
            elif [[ "$fix_layer" == "USECASE" ]]; then
                Use mcp__context7__get-library-docs "/usecase-debugging-solutions" --topic "orchestration-failures" --tokens=3000
            elif [[ "$fix_layer" == "INFRASTRUCTURE" ]]; then
                Use mcp__context7__get-library-docs "/infrastructure-debugging-solutions" --topic "integration-errors" --tokens=3000
            elif [[ "$fix_layer" == "PRESENTATION" ]]; then
                Use mcp__context7__get-library-docs "/presentation-debugging-solutions" --topic "api-errors" --tokens=3000

        **証拠**: 層別テストによる明確な責任特定完了

        1. 特定された問題層のみに集中した最小限修正
        2. 他層への影響を最小化する外科的アプローチ  
        3. 修正後の段階的検証と回帰テスト実行
        
        🆘 **原因が掴めない場合の追加手法**:
        - デバッグログを重要な実行ポイントに追加
        - Context7から最新の解決パターンを取得済み
        - Web検索により類似問題の解決事例を調査済み
        - 段階的なテストケース追加による問題の切り分け
        
        修正を実装しますか？(y/N)"

            Provide specific fix guidance based on:
            - Layer-specific issues identified
            - MCP-3 analysis results
            - Context7 best practices
            - Minimal impact approach
            
            # 修正後検証
            echo "✅ Fix implementation completed - validating results..."
            
            # 修正された層のテスト再実行

                "DOMAIN")
                    execute_domain_focused_debug
                    ;;
                "USECASE")
                    execute_usecase_focused_debug
                    ;;
                "INFRASTRUCTURE") 
                    execute_infra_focused_debug
                    ;;
                "PRESENTATION")
                    execute_presentation_focused_debug
                    ;;
            esac
            
            if [[ $? -eq 0 ]]; then
                echo "✅ Fix validation successful - problem resolved"
                return 0
            else
                echo "❌ Fix validation failed - additional analysis required"
                return 1
            fi
        else
            echo "ℹ️ Fix implementation deferred - debug analysis completed"
            return 0
        fi
    fi
